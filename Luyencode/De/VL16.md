#include <stdio.h>
#include <stdlib.h>

// Hàm tìm ƯCLN sử dụng thuật toán Euclid
long long gcd(long long a, long long b) {
    while (b != 0) {
        long long temp = b;
        b = a % b;
        a = temp;
    }
    return a;
}

// Hàm tính BCNN
long long bcn(long long a, long long b) {
    // Chia trước rồi mới nhân để tránh tràn số (overflow)
    return (a / gcd(a, b)) * b;
}

int main() {
    long long a, b;
    if (scanf("%lld %lld", &a, &b) != 2) return 0;

    long long res = bcn(a, b);
    
    // In ra giá trị tuyệt đối tương ứng với abs() trong Python
    printf("%lld", llabs(res));
    
    return 0;
}
