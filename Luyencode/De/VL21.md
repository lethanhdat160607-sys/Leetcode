#include <stdio.h>

int main() {
    long long N;
    if (scanf("%lld", &N) == 1) {
        long long x = 0;
        long long sum = 0;
        
        // Cộng dồn 1 + 2 + 3 + ... cho đến khi tổng vượt quá N
        while (sum + (x + 1) <= N) {
            x++;
            sum += x;
        }
        
        printf("%lld\n", x);
    }
    return 0;
}
