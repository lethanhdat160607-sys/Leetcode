#include <stdio.h>
#include <stdlib.h>

// Hàm tìm ƯCLN sử dụng thuật toán Euclid
long long gcd(long long a, long long b) {
    while (b != 0) {
        long long temp = b;
        b = a % b;
        a = temp;
    }
    return a;
}

int main() {
    long long a, b;
    if (scanf("%lld %lld", &a, &b) != 2) return 0;
    
    // Mẫu số bằng 0 là không hợp lệ
    if (b == 0) {
        printf("INVALID");
        return 0;
    }
    
    // Nếu tử số bằng 0, giá trị phân số là số nguyên 0
    if (a == 0) {
        printf("0");
        return 0;
    }
    
    // Đảm bảo mẫu số luôn dương để chuẩn hóa dấu
    if (b < 0) {
        a = -a;
        b = -b;
    }
    
    // Tìm ƯCLN để rút gọn
    long long g = gcd(llabs(a), llabs(b));
    a /= g;
    b /= g;
    
    // Nếu mẫu bằng 1 thì in ra dạng số nguyên, ngược lại in dạng phân số tối giản
    if (b == 1) {
        printf("%lld", a);
    } else {
        printf("%lld %lld", a, b);
    }
    
    return 0;
}
