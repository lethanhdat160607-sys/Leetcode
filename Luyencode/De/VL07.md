# include <stdio.h>

long long th(int n, int k){
    if (k > n - k){
        k = n - k;
    }
    long long res =  1; 
    for (int i = 1; i <= k; i++){
        res = res * (n - i + 1) / i;
    }
    return res;
}
int main(){
    int n,k;
    scanf("%d", &n);
    scanf("%d", &k);
    long long res = th(n,k);
    printf("%lld",res);
    return 0;
    
}
