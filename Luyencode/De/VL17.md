#include <stdio.h>
#include <stdlib.h> // Thư viện chứa hàm abs()

int main() {
    int a;
    if (scanf("%d", &a) != 1) return 0;    
    int n = abs(a);
    int dem = 0;
    for (int i = 1; i <= n; i++) {
        if (n % i == 0) {
            dem++;
        }
    }
    printf("%d", dem);
    return 0;
}
