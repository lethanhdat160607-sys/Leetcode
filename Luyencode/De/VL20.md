#include <stdio.h>

int main() {
    char a, b;
    // Nhập 2 ký tự thường cách nhau bởi dấu cách
    scanf("%c %c", &a, &b);
    
    // In các chữ cái viết hoa từ a đến b, cách nhau bằng 1 dấu cách
    for (char i = a; i <= b; i++) {
        // Chuyển ký tự thường sang chữ hoa bằng cách trừ đi 32 (hoặc dùng hàm toupper)
        printf("%c", i - 32);
        if (i < b) {
            printf(" ");
        }
    }
    printf("\n");
    
    return 0;
}
