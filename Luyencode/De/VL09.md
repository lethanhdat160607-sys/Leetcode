#include <stdio.h>

int main() {
    double x;
    int n;
    
    if (scanf("%lf %d", &x, &n) == 2) {
        double S = 0.0;
        double term = 1.0;
        
        for (int i = 1; i <= n; i++) {
            term = term * x / i;
            S += term;
        }
        
        // In làm tròn 2 chữ số thập phân
        printf("%.2f\n", S);
    }
    
    return 0;
}
