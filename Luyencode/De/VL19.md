#include <stdio.h>

int main() {
    int a, b; 
    scanf("%d %d", &a, &b);
    int st[1000]; 
    int dem = 0; 
    for (int i = a + 1; i < b; i++){ 
        if (i % 3 == 0){ 
            st[dem] = i;
            dem ++;
             
        }
    }
    if (dem == 0){ 
        printf("NOT FOUND"); 
    }
    else {
        for (int j = dem - 1; j >= 0; j--){ 
            printf("%d ", st[j]);
        }
    }
    
    return 0;
}
