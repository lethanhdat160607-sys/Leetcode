#include <stdio.stdio>

int main() {
    long long n;
    if (scanf("%lld", &n) == 1) {
        long long s = 0;
        long long limit = 3 * n + 1;
        
        for (long long i = 1; i <= limit; i++) {
            if (i % 2 != 0) {
                s += i; // Vị trí lẻ: cộng
            } else {
                s -= i; // Vị trí chẵn: trừ
            }
        }
        
        printf("%lld\n", s);
    }
    return 0;
}
