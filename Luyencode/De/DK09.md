# include <stdio.h> 


int main(){
    long long n; 
    scanf("%lld", &n); 

    if (n >0 && n <= 100000){   
        if ((n % 100 != 0 && n % 4 == 0) || n % 400 == 0){
            printf("YES");
        } else{
            printf("NO");
        }
    } else{
        printf("INVALID");
    }    

    return 0;
}
