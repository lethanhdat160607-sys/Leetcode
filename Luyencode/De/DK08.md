    int main(){
        float a; 
        float b; 
        char dau;

        scanf("%f", &a); 
        scanf(" %c", &dau);
        scanf("%f", &b);


        if (a <= 10000 && b <= 10000){
            if (dau == '+'){
                printf("%.2f", a + b);
            }
            else if (dau == '-'){
                printf("%.2f", a - b);
            }
            else if (dau == '*'){
                printf("%.2f", a * b);
            }
            else if (dau == '/'){
                if (b == 0){
                    printf("Math Error");
                } else{
                printf("%.2f", a / b);
                }
            }
        } else{
            printf("Math Error");
        }

        return 0;
    }
