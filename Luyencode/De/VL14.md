# include <stdio.h>
# include <stdlib.h>

long long gcd(long long a, long long b) {
    while (b != 0) {
        long long temp = b;
        b = a % b;
        a = temp;
    }
    return a;
}


int main(){
    long long a, b; 
    scanf("%lld %lld ", &a, &b);

    long long res = gcd(llabs(a), llabs(b)); 

    printf("%lld", res);
    

    return 0; 
}
