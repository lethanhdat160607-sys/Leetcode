```
#include <stdio.h>
#include <math.h> // Thư viện chứa hàm round

int main() {
    double n; 
    scanf("%lf", &n);

    // round() sẽ làm tròn đúng chuẩn toán học
    // Dùng (int) để ép kiểu kết quả về số nguyên hiển thị
    printf("%d\n", (int)round(n));

    return 0;
}
```

```
#include <stdio.h>

int main() {
    double n; 
    scanf("%lf", &n);

    int ket_qua;

    if (n >= 0) {
        // Mẹo cho số dương: Cộng 0.5 rồi ép kiểu về int (chặt bỏ phần thập phân)
        ket_qua = (int)(n + 0.5);
    } else {
        // Mẹo cho số âm: Trừ 0.5 rồi ép kiểu về int
        ket_qua = (int)(n - 0.5);
    }

    printf("%d\n", ket_qua);

    return 0;
}
```
