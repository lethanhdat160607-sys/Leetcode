# include <stdio.h> 
# include <math.h>
int main(){

    float a; 
    float b; 
    float c; 

    scanf("%f", &a);
    scanf("%f", &b);
    scanf("%f", &c); 

    if (a == 0){
        if (b == 0){
            if (c == 0){
                printf("WOW");
            }
            else {
                printf("NO");
            }
        }
        else {
            float x = -c / b; 
            if (x == 0) x = 0;
                printf("%.2f", x);
        }
    } else {
        float delta = b * b - 4 * a * c; 
        
        if (delta < 0){ 
            printf("NO");
        } else if (delta == 0){
            double x = -b / (2 * a);
            if (x == 0) x = 0;
            printf("%.2f", x);
        } else {
            float d = sqrt(delta); 
            float x1 = (-b - d) / (2 * a); 
            float x2 = (-b + d) / (2 * a);

            if (x1 > x2) {
                float temp = x1; 
                x1 = x2; 
                x2 = temp; 
            }
            printf("%.2f %.2f", x1, x2);
        }
    }

    return 0; 
}
