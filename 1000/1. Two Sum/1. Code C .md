## Brute Force

int* twoSum(int* nums, int numsSize, int target, int* returnSize) {
    for (int i = 0; i < numsSize; i++) { // duyệt mảng numsSize 
        for (int j = i + 1; j < numsSize; j++) { // duyệt qua tất cả số phía sao i để kiểm tra 
            if (nums[j] == target - nums[i]) { // so sánh lấy vị trí sao i với target khi đã trừ số trước của i 
                int* result = malloc(sizeof(int) * 2); // cấp phát bộ nhớ cho kết quả 
                result[0] = i; // lưu chỉ số của số đầu tiên  
                result[1] = j; // lưu chỉ số của số thứ hai 
                *returnSize = 2; // nếu bộ nhớ bằng 2 thì ok do dề bài kêu chỉ in ra 2 chỉ số 
                return result; // trả về kết quả
            }
        }
    }

    *returnSize = 0; // khi ta đã trả về 2 số rồi thì ta phải cho nó bằng 0 để nó không đọc các file rác từ mảng
    return malloc(sizeof(int) * 0); // trả về một mảng rổng cho hàm chính là 0 byte 
}
