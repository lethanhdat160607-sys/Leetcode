# One-pass Hash Table

```
int* twoSum(int* nums, int numsSize, int target, int* returnSize) {
    struct hashTable {
        int key; // lưu giá trị cần tìm 
        int value; // lưu chỉ số 
        UT_hash_handle hh; // Macro bắt buộc của thư viện quản lý
    } *hashTable = NULL, *item; // tạo rỗng và con trỏ tạm để thêm và xóa bỏ 

    for (int i = 0; i < numsSize; i++) { // duyệt mảng numsSize 
        int complement = target - nums[i]; // tính toán phần bù 
        HASH_FIND_INT(hashTable, &complement, item); // kiểm tra phần bù có trong bảng băm từ vòng lặp đó chưa á 
        if (item) { // nếu item có tồn tại
            int* result = malloc(sizeof(int) * 2); // cấp phát bộ nhớ 
            result[0] = item->value; // lưu chỉ số đầu tiền  
            result[1] = i; // lưu chỉ số thứ hai 
            *returnSize = 2; // thông báo đúng 2 phần tử 
            HASH_CLEAR(hh, hashTable);  // xóa sạch mảng băm 
            return result; // trả về kết quả 
        }
        item = malloc(sizeof(struct hashTable)); // cấp phát động 
        item->key = nums[i]; // lưu giá trị thay đổi của nums[i]
        item->value = i; // lưu lại vị trí 
        HASH_ADD_INT(hashTable, key, item); // thêm cái item vào mảng băm , lấy key là điều kiện mở
    }
    *returnSize = 0; // báo cáo với hệ thông là mảng 0 phần tử 
    HASH_CLEAR(hh, hashTable); // dọn sạch vùng nhớ 
    return malloc(0);  //trả về rỗng để kếp hàm an toàn 
}
```
